\begin{frame}{K-means: nearest centres and Voronoi regions}
  \begin{columns}[T,onlytextwidth]
    \column{.51\textwidth}
    \textbf{Choose $K$, then repeat}\par
    \begin{enumerate}
      \item Assign every cell to its nearest centre.
      \item Update each centre to its cluster mean.
    \end{enumerate}
    \small $K$: number of clusters. Each cell gets one label.\par
    \vspace{2mm}
    \textbf{Geometric model}\par
    Convex regions with linear boundaries.\par
    \textbf{Probabilistic model:} hard-assignment Gaussian clusters with a shared spherical covariance and equal prior weights.
    \column{.45\textwidth}
    \includegraphics[width=\linewidth,height=36mm,keepaspectratio]{../figures/voronoi_lecture.pdf}\par
    \small Yellow points: centres. Each region contains points closest to its centre.
  \end{columns}
  \vspace{2mm}\small\textbf{Our data:} 20,000 live PBMCs, 37 markers, 20 donors.
  \takeaway{Which choice of $K$ resolves meaningful populations, including potentially rare subsets?}
  \source{Figure excerpt: Ulrike von Luxburg, \href{https://www.tml.cs.uni-tuebingen.de/teaching/2024_statistical_learning/downloads_free/vorlesung_main.pdf}{Statistical Machine Learning, Summer 2024}, slide 840 (PDF p. 844), cropped. Gaussian interpretation: Macke / Hennig, Lecture 11.}
\end{frame}

\begin{frame}{Silhouette and Davies--Bouldin: what gets averaged?}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
    \textcolor{Indigo}{\textbf{Silhouette: cell-weighted average}}
    \[\overline{s}=\sum_{c=1}^{K}\frac{n_c}{N}\,\overline{s}_c\]
    \small
    $n_c$: number of cells in cluster $c$.\par
    $N$: total number of cells.\par
    $\overline{s}_c$: mean silhouette in cluster $c$.\par
    \vspace{3mm}
    A group with \textbf{1\,\% of cells} has \textbf{1\,\% weight} for its own mean silhouette.
    \column{.49\textwidth}
    \textcolor{Teal}{\textbf{DB: equal weight per cluster}}
    \[DB=\frac{1}{K}\sum_{c=1}^{K}\max_{j\ne c}
        \frac{S_c+S_j}{\lVert\mu_c-\mu_j\rVert}\]
    \small
    $S_c$: mean distance of cells to their centre.\par
    $\mu_c$: centre (mean vector) of cluster $c$.\par
    \vspace{7mm}
    Each cluster has \textbf{$1/K$ weight}. Lower is better.
  \end{columns}
  \takeaway{DB gives every cluster same weight.}
  \source{Definitions: scikit-learn clustering documentation.}
\end{frame}

\begin{frame}{Subsampling within the observed cells}
  \begin{columns}[T,onlytextwidth]
    \column{.47\textwidth}
    \textbf{Why keep the initial sample?}
    \begin{itemize}
      \item A fresh sample may omit a rare population that we observed initially.
      \item Different cells also prevent a direct membership comparison.
      \item We first test how well observed groups survive removing cells.
    \end{itemize}
    \column{.49\textwidth}
    \textbf{Our stability experiment}
    \begin{enumerate}
      \item Keep 80\,\% of cells within each donor, without replacement.
      \item Refit scaling and clustering with the same parameter settings.
      \item Compare both partitions on the \textbf{same retained cell IDs}.
    \end{enumerate}
    \small Ten repeats, 16,000 cells each, with changing random seeds.
  \end{columns}
  \takeaway{This measures stability conditional on our initial sample. Even 80\,\% subsampling can lose a tiny group.}
  \source{Our notebook implementation.}
\end{frame}

\begin{frame}{Stability: global ARI or cluster-wise Jaccard?}
  \begin{columns}[T,onlytextwidth]
    \column{.43\textwidth}
    \textbf{ARI compares entire partitions}\par
    \small It measures agreement over cell pairs and adjusts for chance.\par
    \vspace{3mm}
    A group of $n$ cells contains $n(n-1)/2$ within-group pairs. Large groups therefore contribute far more pairs.\par
    \vspace{2mm}
    Splitting a rare group changes relatively few pairs, so global ARI can remain high.
    \column{.53\textwidth}
    \textcolor{Teal}{\textbf{Jaccard follows each population}}
    \[J(C,D)=\frac{|C\cap D|}{|C\cup D|}\]
    \small $C$: reference cluster among the retained cells.\par
    $D$: cluster from the new fit on those same cells.\par
  \end{columns}
  \takeaway{Our choice: Jaccard per cluster, then equal cluster weights. Both approaches work without ground truth.}
  \source{SOURCE}
\end{frame}

\begin{frame}{Agglomerative clustering: which groups merge?}
  \begin{columns}[T,onlytextwidth]
    \column{.58\textwidth}
    \includegraphics[width=\linewidth,height=46mm,keepaspectratio]{../figures/linkage_lecture.pdf}\par
    \small $C,C'$: two clusters; $x,y$: cells.\par
    $d(x,y)$: Euclidean distance; $|C|$: cell count.
    \column{.38\textwidth}
    \textbf{Start with one cluster per cell}\par
    \small Merge the closest groups repeatedly. Cut the resulting tree at the desired cluster count.\par
    \vspace{3mm}
    \textbf{Ward uses a different merge rule}\par
    Choose the smallest increase in within-cluster squared error (very k-means like). This favours compact groups.\par
    \vspace{3mm}
    Earlier merges remain fixed in every variant.
  \end{columns}
  \source{Figure excerpt: Ulrike von Luxburg, \href{https://www.tml.cs.uni-tuebingen.de/teaching/2024_statistical_learning/downloads_free/vorlesung_main.pdf}{Statistical Machine Learning, Summer 2024}, slide 850 (PDF p. 854), cropped. Ward: SciPy linkage documentation.}
\end{frame}

\begin{frame}{SKIP: Leiden 1/2: the graph and the objective}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \textbf{Build a graph from marker profiles}
    \begin{itemize}
      \item Each cell becomes a node.
      \item Find neighbouring cells in the 37-dimensional marker space.
      \item Weighted edges represent local similarity between cells.
    \end{itemize}
    \small Neighbour count controls the scale of the graph. Our selected graph uses 30 neighbours.
    \column{.48\textwidth}
    \textbf{Find communities in that graph}\par
    Leiden searches for a partition with a high graph-quality score. \par\vspace{3mm}
    In our setup, the score rewards more internal connection weight than expected from node connectivity.\par\vspace{3mm}
    \small Resolution controls the penalty for the expected connections. Higher values usually yield smaller communities.
  \end{columns}
  \takeaway{We set graph and resolution parameters; the number of clusters emerges. Here: resolution 1.2, 18 groups.}
  \source{Traag, Waltman \& van Eck (2019), pp. 1--2; our notebook uses Scanpy / leidenalg. Graph construction precedes Leiden. The selected objective is the configuration-model (RB) objective, not CPM.}
\end{frame}

\begin{frame}{SKIP: Leiden 2/2: move, refine and aggregate}
  \begin{columns}[T,onlytextwidth]
    \column{.56\textwidth}
    \includegraphics[width=\linewidth,height=47mm,keepaspectratio]{../figures/leiden_algorithm.pdf}
    \column{.41\textwidth}\small
    \textbf{1. Move nodes (a--b)}\par
    Start with one group per node. Move nodes when this improves the quality score.\par\vspace{2mm}
    \textbf{2. Refine groups (b--c)}\par
    Within each group, rebuild connected subgroups. Weakly joined parts can stay separate.\par\vspace{2mm}
    \textbf{3. Aggregate (c--d)}\par
    Each refined subgroup becomes a node. Repeat on the smaller graph (d--f).
  \end{columns}
  \takeaway{Refinement keeps subgroups available for later moves, instead of permanently collapsing an entire group. Improvement from Louvain.}
  \source{Figure 3, cropped: Traag, Waltman \& van Eck (2019), \href{https://doi.org/10.1038/s41598-019-41695-z}{From Louvain to Leiden}, \href{https://creativecommons.org/licenses/by/4.0/}{CC BY 4.0}.}
\end{frame}

\begin{frame}{Backup: K-means initialisation and Ward merges}
  \begin{columns}[T,onlytextwidth]
    \column{.49\textwidth}
    \textbf{K-means++ and repeated starts}
    \small
    \begin{itemize}
      \item Standard k-means++ samples centres in proportion to squared distance from existing centres.
      \item scikit-learn uses a greedy variant with several candidates at each step.
      \item Our \texttt{n\_init=10} retains the run with the lowest inertia.
      \item A fixed seed reproduces a run. Our resampling also changes the seed.
    \end{itemize}
    \column{.47\textwidth}
    \textbf{Ward minimises the increase in SSE}
    \[
      \Delta(A,B)=\frac{n_A n_B}{n_A+n_B}
      \lVert\mu_A-\mu_B\rVert^2
    \]
    \small Ward prefers merges that add little within-cluster variation.\par
    \vspace{3mm}
    $n_A,n_B$: numbers of cells in groups $A,B$. $\mu_A,\mu_B$: their mean vectors. $\Delta$: added squared error.\par\vspace{2mm} Ward only merges. K-means can reassign cells.
  \end{columns}
  \source{scikit-learn KMeans and SciPy linkage documentation. SSE = sum of squared errors.}
\end{frame}

\begin{frame}{Backup: geometry and the singleton example}
  \begin{columns}[T,onlytextwidth]
    \column{.49\textwidth}
    \textbf{Silhouette}
    \[
      s(i)=\frac{b(i)-a(i)}{\max\{a(i),b(i)\}}
    \]
    \small $a(i)$: mean distance to other cells in its cluster.\par
    $b(i)$: smallest mean distance to another cluster.\par
    \vspace{2mm}
    Resolving a nearby subgroup can reduce silhouette even if the subgroup is biologically meaningful. This is a possibility, not an observed comparison in our run.
    \column{.47\textwidth}
    \textbf{Davies--Bouldin with cluster and singleton}
    \[
      S_c=\frac{1}{n_c}\sum_{x_i\in C_c}\lVert x_i-\mu_c\rVert
    \]
    \[
      DB=\frac{S_{\rm large}+0}{\lVert\mu_{\rm large}-x_{\rm singleton}\rVert}
    \]
    \small $C_c$: cluster $c$; $n_c$: its cell count. $x_i$: a cell vector; $\mu_c$: the cluster mean.\par A distant singleton has zero scatter and can yield a low DB.
  \end{columns}
  \source{scikit-learn silhouette and Davies--Bouldin definitions. No claim that silhouette always selects too few clusters, or that DB is neutral across cluster shapes.}
\end{frame}

\begin{frame}{Backup: from K-means loss to Gaussian likelihood}
  \small\setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{2pt}\setlength{\jot}{1pt}
  \begin{align*}
    (r,m)
    &=\arg\min_{r,m}\sum_{i=1}^{N}\sum_{k=1}^{K}r_{ik}\lVert x_i-m_k\rVert^2\\[1mm]
    &=\arg\max_{r,m}\left[\sum_{i=1}^{N}\sum_{k=1}^{K}r_{ik}
       \left(-\frac{\lVert x_i-m_k\rVert^2}{2\sigma^2}\right)+\mathrm{const.}\right]\\[1mm]
    &=\arg\max_{r,m}\prod_{i=1}^{N}\sum_{k=1}^{K}r_{ik}
       \frac{\exp\!\left(-\lVert x_i-m_k\rVert^2/(2\sigma^2)\right)}{Z}\\[1mm]
    &=\arg\max_{r,m}\prod_{i=1}^{N}\sum_{k=1}^{K}r_{ik}\mathcal{N}(x_i;m_k,\sigma^2 I_d)\\[1mm]
    &=\arg\max_{r,m}p(X\mid m,r).
  \end{align*}
  \source{Adapted from Macke / Hennig, Probabilistic Machine Learning (2024), Lecture 11, slide 15, \href{https://creativecommons.org/licenses/by-nc-sa/3.0/}{CC BY-NC-SA 3.0}. Binary one-hot $r$; fixed shared $\sigma^2>0$; $Z=(2\pi\sigma^2)^{d/2}$. Component index corrected to $m_k$.}
\end{frame}

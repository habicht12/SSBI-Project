    \begin{column}{.43\textwidth}
      \eyebrow{One score per donor}
      \[k_d=\lceil0.01N_d\rceil,\qquad
        s_d=\frac{1}{k_d}\sum_{i\in I_d}m_i\]
      {\scriptsize $\boldsymbol{d}$: donor; $\boldsymbol{N_d}$: its cell count; $\boldsymbol{k_d}$: count rounded up; $\boldsymbol{I_d}$: its top-scoring cell indices; $\boldsymbol{i}$: cell; $\boldsymbol{m_i}$: cell margin; $\boldsymbol{s_d}$: mean donor score; $\boldsymbol{\tau}$: donor threshold.}\par\smallskip

        \item Select loss weight $C\in\{0.01,0.1,1\}$ by inner donor ROC-AUC.
        \item Learn $\tau$ from inner out-of-fold donor scores.
        \item Refit on 14 training donors; score all test cells.
      \end{enumerate}
    \end{column}
  \end{columns}

\begin{frame}{Backup: SVM approach 1 --- training the cell scorer}
  \small\compactmath
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.45\textwidth}
      \eyebrow{Training data and scaling}
      \small\compactmath Sample 10,000 cells per training donor without replacement, with fixed seeds. Each cell inherits its donor's label. Fit a shared scaler on these training cells only.
      \[a_{ij}=\operatorname{asinh}(x_{ij}/5),\qquad
        z_{ij}=\frac{a_{ij}-\mu_j}{\sigma_j}\]
      {\scriptsize $\boldsymbol{i}$: cell; $\boldsymbol{j}$: marker; $\boldsymbol{x_{ij}}$: raw value; $\boldsymbol{a_{ij}}$: transformed value; 5: cofactor; $\boldsymbol{\mu_j,\sigma_j}$: training mean and standard deviation; $\boldsymbol{z_{ij}}$: standardized value.}\par\medskip
      \small Equal cell counts give donors equal weight in the training loss. No extra class weighting.\par\smallskip
      {\scriptsize Labels provide weak cell supervision. The 37 markers yield 37 learned weights and one intercept.}
    \end{column}
    \begin{column}{.51\textwidth}
      \eyebrow{Linear score and squared-hinge loss}
      \[m_i=\sum\nolimits_{j=1}^{37}w_jz_{ij}+b\]
      \[\min\nolimits_{w,b}\left[\tfrac12(\|w\|^2+b^2)
        +C\sum\nolimits_{i=1}^{N}\max(0,1-y_i m_i)^2\right]\]
      {\scriptsize $\boldsymbol{w}$: vector of marker weights $w_j$; $\boldsymbol{b}$: intercept; $\boldsymbol{m_i}$: cell margin; $\boldsymbol{N}$: training-cell count; $\boldsymbol{y_i}$: donor label, $-1$ for CMV$-$ and $+1$ for CMV$+$.}\par\smallskip
      {\scriptsize $\boldsymbol{C}$: loss weight; $\boldsymbol{\|w\|^2}$: sum of squared weights; $\boldsymbol{\min_{w,b}}$: optimize weights and intercept. The squared maximum is the squared-hinge loss.}\par\smallskip
      \small Smaller $\boldsymbol{C}$: more regularization. Larger $\boldsymbol{C}$: more weight on fitting labels.\par\smallskip
      {\scriptsize \texttt{LinearSVC} with \texttt{intercept\_scaling=1} also penalizes $b$. No probability calibration.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backup: SVM approach 2 --- model selection and test}
  \small\compactmath
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \eyebrow{1. Choose settings within 14 donors}
      \small Use 3 inner donor folds. Test loss weights $C\in\{0.01,0.1,1\}$ on identical cell samples and scalers within each fold.\par\smallskip
      Highest mean inner donor ROC-AUC wins; ties favour smaller $C$. For that $C$, collect one out-of-fold score per training donor.
      \[J(\tau)=\operatorname{sens}(\tau)+\operatorname{spec}(\tau)-1\]
      {\scriptsize $\boldsymbol{\tau}$: candidate donor threshold; $\boldsymbol{J}$: Youden index; $\boldsymbol{\operatorname{sens}}$: fraction of CMV$+$ donors predicted positive; $\boldsymbol{\operatorname{spec}}$: fraction of CMV$-$ donors predicted negative.}\par\smallskip
      \small Maximize $J$ over finite ROC thresholds; ties choose the highest offered optimum.
    \end{column}
    \begin{column}{.48\textwidth}
      \eyebrow{2. Refit, then evaluate six test donors}
      \small Resample 10,000 cells from each of the 14 training donors; fit a new scaler and SVM. Keep the inner-selected threshold.
      \[k_d=\lceil0.01N_d\rceil,\qquad
        s_d=\frac{1}{k_d}\sum_{i\in I_d}m_i\]
      {\scriptsize $\boldsymbol{d}$: donor; $\boldsymbol{N_d}$: its full cell count; $\boldsymbol{k_d}$: count rounded up; $\boldsymbol{I_d}$: top-scoring cell indices; $\boldsymbol{i}$: cell; $\boldsymbol{m_i}$: margin; $\boldsymbol{s_d}$: donor score.}\par\smallskip
      \small Predict CMV$+$ when $s_d\geq\tau$. AUC uses continuous scores; balanced accuracy uses thresholded labels.\par\medskip
      {\scriptsize The uncalibrated threshold transfers from inner models to a refitted model. Changes in score scale can affect thresholded predictions. Outer test donors choose neither $C$ nor $\tau$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backup: CellCNN subset selection in detail}
  \small\compactmath
  {\small All quantities refer to one saved model and its original training reference.}\par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \eyebrow{Response and reference maximum}
      \[r_{if}=\max(0,w_f^\top z_i+b_f)\]
      \[M_f=\max_{\ell\in T}r_{\ell f},\qquad q_{if}=r_{if}/M_f\]
      {\scriptsize $\boldsymbol{i,\ell}$: cells; $\boldsymbol{f}$: filter; $\boldsymbol{z_i}$: 37 markers standardized by the saved scaler; $\boldsymbol{w_f}$: weights; $\boldsymbol{b_f}$: bias; $\boldsymbol{r_{if}}$: ReLU response; $\boldsymbol{w_f^\top z_i}$: weighted sum.}\par\smallskip
      {\scriptsize $\boldsymbol{T}$: reference cells; $\boldsymbol{M_f}$: maximum response; $\boldsymbol{q_{if}}$: normalized score, defined for $M_f>0$.}\par\medskip
      \small Reconstruct \textbf{20,000 cells per actual training donor} with the original seed: 9 or 10 donors, hence 180,000 or 200,000 reference cells.
    \end{column}
    \begin{column}{.48\textwidth}
      \eyebrow{Direction, subset and centroid}
      \[\Delta_f=v_{f,+}-v_{f,-}\]
      \[S_f=\{i\in T:q_{if}>0.5\},\qquad
        c_f=\frac{1}{|S_f|}\sum_{i\in S_f}a_i\]
      {\scriptsize $\boldsymbol{v_{f,+},v_{f,-}}$: weights from filter $f$ to the CMV$+$/CMV$-$ outputs; $\boldsymbol{\Delta_f}$: output contrast. Its sign gives the direction.}\par\smallskip
      {\scriptsize $\boldsymbol{S_f}$: selected reference cells; $\boldsymbol{|S_f|}$: their count; $\boldsymbol{a_i}$: ArcSinh marker vector; $\boldsymbol{c_f}$: mean marker vector (centroid).}\par\medskip
      \small Skip filters with zero maximum or zero output contrast. Equality at half-maximum is excluded.\par\smallskip
      {\scriptsize Selected cells have equal weight; donor contributions vary. New cells can score above 1.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backup: Citrus subset selection in detail}
  \small\compactmath
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \eyebrow{Which frequency features matter?}
      \[\eta_d=\beta_0+\sum_j\beta_j h_{dj}\]
      {\scriptsize $\boldsymbol{d}$: donor; $\boldsymbol{j}$: cluster; $\boldsymbol{h_{dj}}$: cluster-frequency feature; $\boldsymbol{\beta_0}$: intercept; $\boldsymbol{\beta_j}$: coefficient; $\boldsymbol{\eta_d}$: linear predictor for CMV$+$.}\par\smallskip
      \small\compactmath \texttt{glmnet} standardizes frequency features internally.
      \[q_j=|\beta_j|>10^{-10}\]
      {\scriptsize $\boldsymbol{q_j}$: absolute coefficient; $\boldsymbol{10^{-10}}$: numerical zero tolerance.}\par\smallskip
      \small Retain every effective cluster. A positive coefficient raises the CMV$+$ predictor as frequency increases, with other features fixed; a negative coefficient lowers it.
    \end{column}
    \begin{column}{.48\textwidth}
      \eyebrow{What the exports preserve}
      \[c_j=\frac{1}{|A_j|}\sum_{i\in A_j}a_i\]
      {\scriptsize $\boldsymbol{A_j}$: training cells belonging to cluster $j$; $\boldsymbol{|A_j|}$: their count; $\boldsymbol{i}$: cell; $\boldsymbol{a_i}$: ArcSinh marker vector; $\boldsymbol{c_j}$: saved cluster centroid.}\par\medskip
      \small Training uses 10,000 cells/donor and retains clusters from 0.05\% of training cells. Parent and child clusters can share cells.\par\medskip
      Stored centroids and coefficients do not preserve the full tree and cell memberships needed to assign arbitrary map cells.\par\smallskip
      {\scriptsize Assigning cells to the nearest centroid would introduce a different selection rule. Null models remain in the 30-split recurrence denominator.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backup: SVM cell selection and recurrence}
  \small\compactmath
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \eyebrow{Exact selection within a test donor}
      \[k_d=\lceil0.01N_d\rceil,\qquad t_{is}=m_{is}-\tau_s\]
      {\scriptsize $\boldsymbol{d}$: donor; $\boldsymbol{s}$: outer split; $\boldsymbol{i}$: cell; $\boldsymbol{N_d}$: full cell count; $\boldsymbol{k_d}$: count rounded up; $\boldsymbol{m_{is}}$: cell margin; $\boldsymbol{\tau_s}$: learned donor threshold; $\boldsymbol{t_{is}}$: centered score.}\par\medskip
      \small Let $\boldsymbol{I_{ds}}$ contain the $k_d$ highest-scoring cell indices. Break boundary ties by original FCS event number.\par\medskip
      Within $\boldsymbol{I_{ds}}$: positive if $t_{is}>0$, negative if $t_{is}<0$, neither if $t_{is}=0$. Cells outside $I_{ds}$ are unselected.\par\medskip
      {\scriptsize Example: 101 cells give 2 top cells. Margins 9 and 8 at threshold 8.5 give one positive and one negative selection.}
    \end{column}
    \begin{column}{.48\textwidth}
      \eyebrow{Repeat across that donor's test visits}
      \[f_i^+=\frac{n_i^+}{|\mathcal T_d|}\]
      {\scriptsize $\boldsymbol{\mathcal T_d}$: all splits testing donor $d$; $\boldsymbol{|\mathcal T_d|}$: their count; $\boldsymbol{n_i^+}$: visits where cell $i$ is in $I_{ds}$ and $t_{is}>0$; $\boldsymbol{f_i^+}$: positive selection frequency.}\par\smallskip
      \small\compactmath Include visits without selection: 3 positive selections in 8 test visits give 3/8.
      \[s_{ds}-\tau_s=\frac{1}{k_d}\sum_{i\in I_{ds}}t_{is}\]
      {\scriptsize $\boldsymbol{s_{ds}}$: mean of the selected cell margins, the donor score in split $s$.}\par\smallskip
      \small Fixed-selection decomposition of the donor score; no causal cell-removal effect.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CellCNN learns filters for informative cells}
  \begin{center}
    \begin{tikzpicture}[x=1mm,y=1mm]
      \node[stage] (a) at (0,0) {\textbf{Cell group}\\3,000 $\times$ 37};
      \node[stage] (b) at (36,0) {\textbf{3--5 filters}\\linear + ReLU};
      \node[stage] (c) at (72,0) {\textbf{Top 1\%}\\mean per filter};
      \node[stage] (d) at (108,0) {\textbf{Softmax}\\CMV probability};
      \draw[flow] (a) -- (b); \draw[flow] (b) -- (c); \draw[flow] (c) -- (d);
    \end{tikzpicture}
  \end{center}
  \vspace{0mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.49\textwidth}
      \eyebrow{Learn which cells matter}
      \[r_{if}=\max(0,w_f^\top z_i+b_f)\]
      A filter scores a marker combination. Pooling focuses on the strongest responses; the whole network learns from donor labels.
    \end{column}
    \begin{column}{.47\textwidth}
      \eyebrow{Our paper-guided implementation}
      \begin{itemize}\setlength\itemsep{2pt}
        \item PyTorch; 200 cell groups per training donor.
        \item Choose the best inner-fold network; no outer refit.
        \item Adaptations: fixed 3/4/5-filter search; average five test-input probabilities.
      \end{itemize}
    \end{column}
  \end{columns}
  \takeaway{Filter learning and donor classification are trained jointly.}
  \source{\papersource}
\end{frame}

\begin{frame}{Citrus predicts from population frequencies}
  \begin{center}
    \begin{tikzpicture}[x=1mm,y=1mm]
      \node[stage] (a) at (0,0) {\textbf{Balanced cells}\\10,000/donor};
      \node[stage] (b) at (36,0) {\textbf{Ward tree}\\training cells};
      \node[stage] (c) at (72,0) {\textbf{Frequencies}\\donor $\times$ cluster};
      \node[stage] (d) at (108,0) {\textbf{L1 logistic}\\CMV probability};
      \draw[flow] (a) -- (b); \draw[flow] (b) -- (c); \draw[flow] (c) -- (d);
    \end{tikzpicture}
  \end{center}
  \vspace{2mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.49\textwidth}
      \eyebrow{First define populations}
      Retain tree clusters containing at least \textbf{0.05\% of training cells}. Parent and child clusters may overlap.\par\medskip
      Each donor is represented by the fraction of its cells in each retained cluster.
    \end{column}
    \begin{column}{.47\textwidth}
      \eyebrow{Then select predictive frequencies}
      Nonzero L1-logistic coefficients identify selected model populations. New cells inherit memberships from their nearest training cell.\par\medskip
      \small Compute compromise: 10,000 cells and 30 splits, versus 20,000 and 100 in the paper.
    \end{column}
  \end{columns}
  \takeaway{Citrus separates population discovery from supervised prediction.}
  \source{\citrussource\ \papersource}
\end{frame}

\begin{frame}{Our adaptation: aggregate the highest cell margins}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.53\textwidth}
      \eyebrow{A few high-scoring cells can matter}
      \begin{tikzpicture}[x=1mm,y=1mm]
        \fill[Pale] (0,0) rectangle (35,13);
        \node[font=\small,align=center] at (17.5,6.5) {990 remaining\\cells};
        \foreach \i in {0,...,9} {
          \fill[Indigo] ({40+2.4*\i},0) rectangle ({41.6+2.4*\i},{10.5+.5*\i});
        }
        \draw[-{Stealth},draw=Muted] (0,-3) -- (65,-3);
        \node[font=\scriptsize,align=center] at (33,-8) {cells sorted by margin $m_i$};
        \node[font=\scriptsize,text=Indigo] at (51,20) {10 highest margins};
      \end{tikzpicture}\par\smallskip
      \small Toy example: 1,000 cells; keep 10.\par
      Margins 2.1, 2.2, \ldots, 3.0 give $s_d=2.55$.\par
      At $\tau=1.80$: predict CMV$+$.
    \end{column}
    \begin{column}{.43\textwidth}
      \eyebrow{One score per donor}
      \[k_d=\lceil0.01N_d\rceil,\qquad
        s_d=\frac{1}{k_d}\sum_{i\in\mathrm{Top}_{k_d}}m_i\]
      \begin{enumerate}\setlength\itemsep{3pt}
        \item Select $C\in\{0.01,0.1,1\}$ by inner donor ROC-AUC.
        \item Learn $\tau$ from inner out-of-fold donor scores.
        \item Refit on 14 training donors; score all test cells.
      \end{enumerate}
    \end{column}
  \end{columns}
  \takeaway{Our project adaptation is the Top-1-\% donor score, fixed before testing.}
\end{frame}

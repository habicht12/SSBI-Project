\begin{frame}{Backup: half-max population frequency}
  % Retain the exact rule, including M=0 and missing positive-output filters.
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.54\textwidth}
      \eyebrow{Filter selection and fixed training threshold}
      \[\Delta_k=V_{1k}-V_{0k},\qquad k^*=\arg\max_{k:\Delta_k>0}\Delta_k\]
      \[M=\max_{x\in\text{inner-train cells}}h_{k^*}(x),\qquad t=\tfrac12M\]
      \[f_i=\frac{\#\{x\in\text{test sample }i:h_{k^*}(x)>t\}}{N_i}\]
      \small One fixed 20,000-cell sample per test donor. The strict rule also applies when $M=0$.
    \end{column}
    \begin{column}{.42\textwidth}
      \eyebrow{Two population metrics}
      \[\mathrm{AUC}_{\mathrm{freq}}=\operatorname{AUC}(y_i,f_i)\]
      \[\mathrm{Effect}=\bar f_{\mathrm{CMV+}}-\bar f_{\mathrm{CMV-}}\]
      \small Without a positive filter, values remain missing. No post-hoc flipping of AUC direction.\par\medskip
      Frequency comparisons use \textbf{\CommonFrequencySplits{}} jointly evaluable splits.
    \end{column}
  \end{columns}
  \takeaway{Half-max is cell selection, not clustering; CMV is a donor label, not a cell-type label.}
  \source{The baseline has no positive-output filter in splits 44 and 84. These two splits are excluded from all common frequency comparisons, but retained for network metrics.}
\end{frame}

\begin{frame}{Backup: scope of the controlled comparison}
  \begin{itemize}
    \item \textbf{Quadratic Hard vs.\ Soft:} same cell response and L2 penalty; the pooling function changes.
    \item \textbf{CellCNN vs.\ Quadratic:} adds a quadratic term and its L2 penalty.
    \item \textbf{Prototype vs.\ Quadratic:} different geometry, initialization and regularization; not an isolated alpha comparison.
    % Add the omitted distinction between outer-training and actual fit donors.
    \item \textbf{No refit:} the selected inner model uses 9--10 training donors, not all 14 outer-training donors.
    \item \textbf{Limitation:} variants were developed after earlier results on these same 20 donors.
  \end{itemize}
  {\small L2 coefficient $10^{-4}$: CellCNN on $W,V$; Quadratic on $W,Q,V$; Prototype on $V$, plus $10^{-3}\operatorname{mean}((a-1)^2)$. No extra alpha or radius penalty.\par}
  \takeaway{\FinalTakeaway}
  \source{Nine candidates per outer split: three inner folds and 3/4/5 filters. The same donor splits, sampling seeds, learning rate and early-stopping procedure are used.}
\end{frame}

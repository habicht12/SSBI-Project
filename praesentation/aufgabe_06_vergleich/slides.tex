\begin{frame}{Take-away and scope}
  \begin{itemize}
    \item Quadratic geometry (hard or soft) gives the best network classification here.
    \item Learnable soft pooling doesn't add further gain: learned $\alpha\approx$ the fixed 1\% cutoff.
    \item Quadratic Soft-$\alpha$ beats CellCNN in 53/100 paired splits (mean $\Delta$AUC $+0.105$).
    \item Limitation: only 20 independent donors -- more splits are not more independent evidence.
  \end{itemize}
  \takeaway{Quadratic Hard and Soft both reach 0.913 mean network AUC; learnable soft pooling doesn't increase it here.}
\end{frame}

\begin{frame}{Backup: selecting a filter and defining half-max population frequency}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.54\textwidth}
      \eyebrow{Filter selection and fixed training threshold}
      \[\Delta_k=V_{1k}-V_{0k},\qquad k^*=\arg\max_{k:\Delta_k>0}\Delta_k\]
      \[M=\max_{x\in\text{inner-train cells}}h_{k^*}(x),\qquad t=\tfrac12M\]
      \[f_i=\frac{\#\{x\in\text{test sample }i:h_{k^*}(x)>t\}}{N_i}\]
      \small Same fixed sample of up to 20{,}000 cells per test donor. The strict rule also applies when $M=0$.
    \end{column}
    \begin{column}{.42\textwidth}
      \eyebrow{Two population metrics}
      \[\mathrm{AUC}_{\mathrm{freq}}=\operatorname{AUC}(y_i,f_i)\]
      \[\mathrm{Effect}=\bar f_{\mathrm{CMV+}}-\bar f_{\mathrm{CMV-}}\]
      \small Without a positive filter, values remain missing. No post-hoc flipping of AUC direction.
      \medskip\par
      For frequency comparisons: \textbf{98} jointly determinable splits.
    \end{column}
  \end{columns}
  \takeaway{Half-max is a cell-selection rule, not clustering; CMV is a donor label, not a cell-type label.}
\end{frame}

\begin{frame}{Backup: what does the extra pooling parameter learn?}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.58\textwidth}
      \includegraphics[width=\linewidth,height=51mm,keepaspectratio]{learned_alphas.pdf}
    \end{column}
    \begin{column}{.39\textwidth}
      \eyebrow{Rank fraction, not population frequency}
      \[\alpha_k=\sigma(\theta_k),\quad \alpha_k^{(0)}=0.01\]
      \[m_{rk}=\sigma\!\left(\frac{\alpha_k-q_r}{0.002}\right)\]
      \small With $q_r=(r-\tfrac12)/N$. Temperature stays fixed. A soft $\alpha$ of 1\% is not identical to hard top-1\%.
    \end{column}
  \end{columns}
  \takeaway{Median $\alpha$: Quadratic 0.96\%; Prototype 1.04\%. Alpha describes pooling, not the measured cell frequency.}
\end{frame}

\begin{frame}{Backup: scope of the controlled comparison}
  \begin{itemize}
    \item \textbf{Quadratic Hard vs.\ Soft:} same cell response and L2 penalty; only the pooling function changes.
    \item \textbf{CellCNN vs.\ Quadratic:} adds a quadratic term and its L2 penalty.
    \item \textbf{Prototype vs.\ Quadratic:} different geometry, initialization and regularization; not an isolated alpha comparison.
    \item \textbf{Limitation:} variants were developed after earlier results. More splits do not create new independent donors.
  \end{itemize}
  \medskip
  \textbf{Regularization:} CellCNN on $W,V$; Quadratic on $W,Q,V$. Prototype: output L2 plus $10^{-3}\operatorname{mean}((a-1)^2)$.
  \takeaway{Quadratic Hard and Soft both reach 0.913 mean network AUC; learnable soft pooling doesn't increase it here.}
  \plotcontext{All 100 planned splits are reported. Code, checkpoints, tables and figures are documented together.}
\end{frame}

\begin{frame}{Eine gemeinsame Half-Max-Definition für alle vier Modelle}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.54\textwidth}
      \eyebrow{Filterauswahl und feste Trainingsschwelle}
      \[\Delta_k=V_{1k}-V_{0k},\qquad k^*=\arg\max_{k:\Delta_k>0}\Delta_k\]
      \[M=\max_{x\in\text{Inner-Train-Zellen}}h_{k^*}(x),\qquad t=\tfrac12M\]
      \[f_i=\frac{\#\{x\in\text{Teststichprobe }i:h_{k^*}(x)>t\}}{N_i}\]
      \small Gleiche feste Stichprobe von bis zu 20.000 Zellen je Testspender. Die strikte Regel gilt auch bei $M=0$.
    \end{column}
    \begin{column}{.42\textwidth}
      \eyebrow{Zwei Populationsmetriken}
      \[\mathrm{AUC}_{\mathrm{freq}}=\operatorname{AUC}(y_i,f_i)\]
      \[\mathrm{Effekt}=\bar f_{\mathrm{CMV+}}-\bar f_{\mathrm{CMV-}}\]
      \small Ohne positiven Filter bleiben die Werte fehlend. Kein nachträgliches Umdrehen der AUC.
      \medskip\par
      Für Frequency-Vergleiche: \textbf{\CommonFrequencySplits{} gemeinsam bestimmbare Splits}.
    \end{column}
  \end{columns}
  \takeaway{Half-Max ist eine Zellselektion. Es wird kein Clustering ausgeführt; CMV ist kein Zelltyp-Label.}
\end{frame}

\begin{frame}{Einordnung des kontrollierten Vergleichs}
  \begin{itemize}
    \item \textbf{Quadratic Hard vs. Soft:} gleiche Zellresponse und L2-Strafe; geändert wird die Poolingfunktion.
    \item \textbf{CellCNN vs. Quadratic:} zusätzlicher quadratischer Term und dessen L2-Strafe.
    \item \textbf{Prototyp vs. Quadratic:} andere Geometrie, Initialisierung und Regularisierung; kein isolierter Alpha-Vergleich.
    \item \textbf{Begrenzung:} Varianten wurden nach früheren Ergebnissen entwickelt. Mehr Splits erzeugen keine neuen unabhängigen Donoren.
  \end{itemize}
  \medskip
  \textbf{Regularisierung:} CellCNN auf $W,V$; Quadratic auf $W,Q,V$.
  Prototyp: Output-L2 plus $10^{-3}\operatorname{mean}((a-1)^2)$.
  \takeaway{\FinalTakeaway}
  \plotcontext{Alle 100 vorgesehenen Splits werden berichtet. Code, Checkpoints, Tabellen und Abbildungen sind gemeinsam dokumentiert.}
\end{frame}

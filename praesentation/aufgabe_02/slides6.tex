\begin{frame}{Eine gemeinsame Half-Max-Definition für alle vier Modelle}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.54\textwidth}
      \eyebrow{Filterauswahl und feste Trainingsschwelle}
      \[\Delta_k=V_{1k}-V_{0k},\qquad k^*=\arg\max_{k:\Delta_k>0}\Delta_k\]
      \[M=\max_{x\in\text{Inner-Train-Zellen}}h_{k^*}(x),\qquad t=\tfrac12M\]
      \[f_i=\frac{\#\{x\in\text{Teststichprobe }i:h_{k^*}(x)>t\}}{N_i}\]
      \small Gleiche feste Stichprobe von bis zu 20.000 Zellen je Testspender. Die strikte Regel gilt auch bei $M=0$.
    \end{column}
    \begin{column}{.42\textwidth}
      \eyebrow{Zwei Populationsmetriken}
      \[\mathrm{AUC}_{\mathrm{freq}}=\operatorname{AUC}(y_i,f_i)\]
      \[\mathrm{Effekt}=\bar f_{\mathrm{CMV+}}-\bar f_{\mathrm{CMV-}}\]
      \small Ohne positiven Filter bleiben die Werte fehlend. Kein nachträgliches Umdrehen der AUC.
      \medskip\par
      Für Frequency-Vergleiche: \textbf{\CommonFrequencySplits{} gemeinsam bestimmbare Splits}.
    \end{column}
  \end{columns}
  \takeaway{Half-Max ist eine Zellselektion. Es wird kein Clustering ausgeführt; CMV ist kein Zelltyp-Label.}
\end{frame}

\begin{frame}{Einordnung des kontrollierten Vergleichs}
  \begin{itemize}
    \item \textbf{Quadratic Hard vs. Soft:} gleiche Zellresponse und L2-Strafe; geändert wird die Poolingfunktion.
    \item \textbf{CellCNN vs. Quadratic:} zusätzlicher quadratischer Term und dessen L2-Strafe.
    \item \textbf{Prototyp vs. Quadratic:} andere Geometrie, Initialisierung und Regularisierung; kein isolierter Alpha-Vergleich.
    \item \textbf{Begrenzung:} Varianten wurden nach früheren Ergebnissen entwickelt. Mehr Splits erzeugen keine neuen unabhängigen Donoren.
  \end{itemize}
  \medskip
  \textbf{Regularisierung:} CellCNN auf $W,V$; Quadratic auf $W,Q,V$.
  Prototyp: Output-L2 plus $10^{-3}\operatorname{mean}((a-1)^2)$.
  \takeaway{\FinalTakeaway}
  \plotcontext{Alle 100 vorgesehenen Splits werden berichtet. Code, Checkpoints, Tabellen und Abbildungen sind gemeinsam dokumentiert.}
\end{frame}
